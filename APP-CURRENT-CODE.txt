function normalizeStatus(value: string) {
  return value
    .trim()
    .toLowerCase()
    .replace(/[\s-]+/g, "_");
}

function formatLabel(value: string) {
  return value
    .replace(/[_-]+/g, " ")
    .replace(/\b\w/g, (letter) =>
      letter.toUpperCase()
    );
}

function formatDate(value: string | null) {
  if (!value) {
    return "Date unavailable";
  }

  try {
    return new Intl.DateTimeFormat("en-IN", {
      dateStyle: "medium",
      timeStyle: "short",
    }).format(new Date(value));
  } catch {
    return value;
  }
}

function StatusBadge({
  status,
}: {
  status: string;
}) {
  const normalized =
    normalizeStatus(status);

  if (normalized === "pending") {
    return (
      <span className="inline-flex rounded-full bg-amber-50 px-2.5 py-1 text-[10px] font-extrabold uppercase text-amber-700">
        PENDING
      </span>
    );
  }

  if (normalized === "approved") {
    return (
      <span className="inline-flex rounded-full bg-blue-50 px-2.5 py-1 text-[10px] font-extrabold uppercase text-blue-700">
        APPROVED
      </span>
    );
  }

  if (normalized === "published") {
    return (
      <span className="inline-flex rounded-full bg-emerald-50 px-2.5 py-1 text-[10px] font-extrabold uppercase text-emerald-700">
        PUBLISHED
      </span>
    );
  }

  if (normalized === "rejected") {
    return (
      <span className="inline-flex rounded-full bg-red-50 px-2.5 py-1 text-[10px] font-extrabold uppercase text-red-700">
        REJECTED
      </span>
    );
  }

  if (normalized === "duplicate") {
    return (
      <span className="inline-flex rounded-full bg-slate-100 px-2.5 py-1 text-[10px] font-extrabold uppercase text-slate-600">
        DUPLICATE
      </span>
    );
  }

  return (
    <span className="inline-flex rounded-full bg-slate-100 px-2.5 py-1 text-[10px] font-extrabold uppercase text-slate-600">
      {formatLabel(status)}
    </span>
  );
}

function createAdminClient() {
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const serviceRoleKey = process.env.SUPABASE_SERVICE_ROLE_KEY;

  if (!url || !serviceRoleKey) {
    throw new Error(
      "Supabase environment variables are missing."
    );
  }

  return createClient(url, serviceRoleKey, {
    auth: {
      autoRefreshToken: false,
      persistSession: false,
    },
  });
}

function formatDate(value: string | null) {
  if (!value) return "â€”";

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) {
    return value;
  }

  return new Intl.DateTimeFormat("en-IN", {
    day: "2-digit",
    month: "short",
    year: "numeric",
  }).format(date);
}

function isExpired(job: Job) {
  if (!job.expiry_date) return false;

  const expiry = new Date(job.expiry_date);

  if (Number.isNaN(expiry.getTime())) {
    return false;
  }

  return expiry.getTime() < Date.now();
}

function getStatus(job: Job) {
  if (isExpired(job)) {
    return {
      label: "Expired",
      className:
        "bg-red-50 text-red-700 ring-1 ring-red-100",
    };
  }

  if (job.is_published) {
    return {
      label: "Published",
      className:
        "bg-emerald-50 text-emerald-700 ring-1 ring-emerald-100",
    };
  }

  return {
    label: "Draft",
    className:
      "bg-amber-50 text-amber-700 ring-1 ring-amber-100",
  };
}

function createAdminClient() {
  const url = process.env.NEXT_PUBLIC_SUPABASE_URL;
  const serviceRoleKey = process.env.SUPABASE_SERVICE_ROLE_KEY;

  if (!url || !serviceRoleKey) {
    throw new Error(
      "Supabase environment variables are missing."
    );
  }

  return createClient(url, serviceRoleKey, {
    auth: {
      autoRefreshToken: false,
      persistSession: false,
    },
  });
}

function textValue(
  formData: FormData,
  name: string
) {
  const value = formData.get(name);

  return typeof value === "string"
    ? value.trim()
    : "";
}

function nullableText(
  formData: FormData,
  name: string
) {
  const value = textValue(formData, name);

  return value || null;
}

function nullableNumber(
  formData: FormData,
  name: string
) {
  const value = textValue(formData, name);

  if (!value) return null;

  const number = Number(value);

  return Number.isFinite(number)
    ? number
    : null;
}

function nullableDate(
  formData: FormData,
  name: string
) {
  const value = textValue(formData, name);

  return value || null;
}

async function createJob(
  formData: FormData
) {
  "use server";

  const user = await getCurrentUser();

  if (!user) {
    redirect("/login?next=/admin/jobs/new");
  }

  const admin = await getCurrentAdmin();

  if (!admin) {
    redirect("/profile");
  }

  const title = textValue(formData, "title");
  const organization = textValue(
    formData,
    "organization"
  );

  if (!title) {
    throw new Error("Job title is required.");
  }

  if (!organization) {
    throw new Error(
      "Organization name is required."
    );
  }

  const supabase = createAdminClient();

  const job = {
    title,
    organization,
    location: nullableText(formData, "location"),
    qualification: nullableText(
      formData,
      "qualification"
    ),
    type: nullableText(formData, "type"),
    category: nullableText(
      formData,
      "category"
    ),
    description: nullableText(
      formData,
      "description"
    ),
    vacancies: nullableNumber(
      formData,
      "vacancies"
    ),
    age_limit: nullableText(
      formData,
      "age_limit"
    ),
    salary: nullableText(
      formData,
      "salary"
    ),
    application_fee: nullableText(
      formData,
      "application_fee"
    ),
    job_type: nullableText(
      formData,
      "job_type"
    ),
    application_start: nullableDate(
      formData,
      "application_start"
    ),
    last_date: nullableDate(
      formData,
      "last_date"
    ),
    exam_date: nullableDate(
      formData,
      "exam_date"
    ),
    official_website: nullableText(
      formData,
      "official_website"
    ),
    notification_url: nullableText(
      formData,
      "notification_url"
    ),
    apply_url: nullableText(
      formData,
      "apply_url"
    ),
    expiry_date: nullableDate(
      formData,
      "expiry_date"
    ),
    is_published:
      formData.get("is_published") === "on",
  };

  const { error } = await supabase
    .from("jobs")
    .insert(job);

  if (error) {
    throw new Error(
      `Unable to create job: ${error.message}`
    );
  }

  revalidatePath("/jobs");
  revalidatePath("/admin/jobs");
  revalidatePath("/");

  redirect("/admin/jobs");
}

    event.preventDefault();

    setError("");
    setLoading(true);

    const cleanEmail = email.trim().toLowerCase();

    if (!cleanEmail || !password) {
      setError("Please enter your email and password.");
      setLoading(false);
      return;
    }

    try {
      const supabase = createClient();

      const { data, error: loginError } =
        await supabase.auth.signInWithPassword({
          email: cleanEmail,
          password,
        });

      if (loginError) {
        setError(loginError.message || "Invalid email or password.");
        setLoading(false);
        return;
      }

      if (!data.user) {
        setError("Login could not be completed.");
        setLoading(false);
        return;
      }

      /*
       * If the user originally tried to open a protected page,
       * send them back there.
       */
      if (
        next &&
        next.startsWith("/") &&
        !next.startsWith("//")
      ) {
        router.replace(next);
        router.refresh();
        return;

       * Ask our server whether the current authenticated user
       * is an administrator.
       */
      const response = await fetch("/api/auth/redirect", {
        method: "GET",
        cache: "no-store",
      });

        const result = await response.json();

          router.replace("/admin");
          router.refresh();
          return;
        }

      /*
       * Normal authenticated users go to Profile.

async function toggleRead(formData: FormData) {

    redirect("/login?next=/admin/notifications");

  if (!admin) {

  }

  if (!id) {

    .update({
      is_read: current !== "true",

    .eq("id", id);

  revalidatePath("/admin/notifications");
}

async function saveSettings(formData: FormData) {
  "use server";

  const user = await getCurrentUser();

  if (!user) {
    redirect("/login?next=/admin/settings");
  }

  const admin = await getCurrentAdmin();

  if (!admin) {
    redirect("/profile");
  }

  const siteName =
    String(formData.get("site_name") || "").trim();

  const siteUrl =
    String(formData.get("site_url") || "").trim();

  const siteDescription =
    String(formData.get("site_description") || "").trim();

  const contactEmail =
    String(formData.get("contact_email") || "").trim();

  const maintenanceMode =
    formData.get("maintenance_mode") === "on";

  const maintenanceMessage =
    String(
      formData.get("maintenance_message") || ""
    ).trim();

  if (!siteName || !siteUrl || !siteDescription) {
    redirect(
      "/admin/settings?error=Site name, URL and description are required."
    );
  }

  const supabase = createAdminClient();

  const settings = [
    {
      setting_key: "site_name",
      setting_value: siteName,
    },
    {
      setting_key: "site_url",
      setting_value: siteUrl,
    },
    {
      setting_key: "site_description",
      setting_value: siteDescription,
    },
    {
      setting_key: "contact_email",
      setting_value: contactEmail,
    },
    {
      setting_key: "maintenance_mode",
      setting_value: maintenanceMode ? "true" : "false",
    },
    {
      setting_key: "maintenance_message",
      setting_value:
        maintenanceMessage ||
        "CareerPortal is currently undergoing maintenance. Please check back soon.",
    },
  ];

  for (const setting of settings) {
    const { data: existing, error: findError } =
      await supabase
        .from("site_settings")
        .select("id")
        .eq("setting_key", setting.setting_key)
        .maybeSingle();

    if (findError) {
      console.error(
        "Settings lookup error:",
        findError
      );

      redirect(
        `/admin/settings?error=${encodeURIComponent(
          findError.message
        )}`
      );
    }

    if (existing?.id) {
      const { error: updateError } =
        await supabase
          .from("site_settings")
          .update({
            setting_value: setting.setting_value,
            updated_at: new Date().toISOString(),
          })
          .eq("id", existing.id);

      if (updateError) {
        console.error(
          "Settings update error:",
          updateError
        );

        redirect(
          `/admin/settings?error=${encodeURIComponent(
            updateError.message
          )}`
        );
      }
    } else {
      const { error: insertError } =
        await supabase
          .from("site_settings")
          .insert(setting);

      if (insertError) {
        console.error(
          "Settings insert error:",
          insertError
        );

        redirect(
          `/admin/settings?error=${encodeURIComponent(
            insertError.message
          )}`
        );
      }
    }
  }

  revalidatePath("/");
  revalidatePath("/admin");
  revalidatePath("/admin/settings");
  revalidatePath("/maintenance");

  redirect("/admin/settings?saved=true");
}

function getSetting(
  settings: Array<{
    setting_key: string;
    setting_value: string | null;
  }>,
  key: string,
  fallback = ""
) {
  return (
    settings.find(
      (setting) => setting.setting_key === key
    )?.setting_value ?? fallback
  );
}

function formatDate(value: string | null) {

    return "Unavailable";

      timeStyle: "short",
    }).format(new Date(value));
  } catch {
    return value;
  }
}

function toNumber(
  value: unknown
): number | null {
  if (
    value === null ||
    value === undefined ||
    value === ""
  ) {
    return null;
  }

  const number = Number(value);

  return Number.isFinite(number)
    ? number
    : null;
}

function normalizeStatus(
  value: string | null
) {
  return String(value || "")
    .trim()
    .toLowerCase()
    .replace(/[\s-]+/g, "_");
}

function StatusBadge({
  status,
}: {
  status: string | null;
}) {
  const normalized =
    normalizeStatus(status);

  if (normalized === "success") {
    return (
      <span className="inline-flex rounded-full bg-emerald-50 px-2.5 py-1 text-[10px] font-extrabold uppercase text-emerald-700">
        SUCCESS
      </span>
    );
  }

  if (normalized === "failed" || normalized === "error") {
    return (
      <span className="inline-flex rounded-full bg-red-50 px-2.5 py-1 text-[10px] font-extrabold uppercase text-red-700">
        FAILED
      </span>
    );
  }

  if (normalized === "running") {
    return (
      <span className="inline-flex rounded-full bg-amber-50 px-2.5 py-1 text-[10px] font-extrabold uppercase text-amber-700">
        RUNNING
      </span>
    );
  }

  return (
    <span className="inline-flex rounded-full bg-slate-100 px-2.5 py-1 text-[10px] font-extrabold uppercase text-slate-600">
      {status || "UNKNOWN"}
    </span>
  );
}

function StatusPill({
  label,
  value,
  type,
}: {
  label: string;
  value: number;
  type: "success" | "failed" | "running";
}) {
  const styles = {
    success:
      "bg-emerald-50 text-emerald-700",
    failed:
      "bg-red-50 text-red-700",
    running:
      "bg-amber-50 text-amber-700",
  };

  return (
    <span
      className={`rounded-full px-3 py-2 text-xs font-extrabold ${styles[type]}`}
    >
      {label}: {value}
    </span>
  );
}

function MiniStat({
  label,
  value,
}: {
  label: string;
  value: number;
}) {
  return (
    <div className="rounded-xl bg-slate-50 p-3 text-center">

        {label}
      </p>

      <p className="mt-1 text-sm font-extrabold text-slate-800">
        {value}
      </p>
    </div>
  );
}

function StatCard({

function formatDate(value: string | null) {
  if (!value) {
    return "Date unavailable";
  }

  try {
    return new Intl.DateTimeFormat("en-IN", {
      dateStyle: "medium",
      timeStyle: "short",
    }).format(new Date(value));
  } catch {
    return value;
  }
}

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) return value;

    day: "2-digit",
    month: "short",
    year: "numeric",
  });
}

function formatDate(value: string | null) {
  if (!value) return "Not specified";

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) return value;

  return date.toLocaleDateString("en-IN", {
    day: "2-digit",
    month: "short",
    year: "numeric",
  });
}

function formatDate(value: string | null) {
  if (!value) return "Not specified";

  const date = new Date(value);

  if (Number.isNaN(date.getTime())) return value;

  return date.toLocaleDateString("en-IN", {
    day: "2-digit",
    month: "short",
    year: "numeric",
  });
}

  const [password, setPassword] = useState("");

  const [loading, setLoading] = useState(false);
  const [errorMessage, setErrorMessage] = useState("");

  async function handleLogin(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();

    if (loading) return;

    setLoading(true);
    setErrorMessage("");

    try {
      const cleanEmail = email.trim().toLowerCase();

      if (!cleanEmail || !password) {
        setErrorMessage("Email aur password dono enter karein.");
        setLoading(false);
        return;
      }

      // 1. Login
      const { data, error } = await supabase.auth.signInWithPassword({
        email: cleanEmail,
        password,
      });

      if (error) {
        setErrorMessage(error.message);
        setLoading(false);
        return;
      }

      if (!data.user || !data.session) {
        setErrorMessage("Login session create nahi ho saka.");
        setLoading(false);
        return;
      }

      // 2. Give Supabase a moment to finish browser session storage.
      await new Promise((resolve) => setTimeout(resolve, 300));

      // 3. Ask server where this user should go.
      const response = await fetch("/api/auth/redirect", {
        method: "GET",
        cache: "no-store",
        credentials: "include",
      });

      const destinationData = await response.json();

      // 4. If ?next= exists, respect it only for non-admin users.
      const next = searchParams.get("next");

      if (
        next &&
        next.startsWith("/") &&
        !next.startsWith("//") &&
        !destinationData.isAdmin
      ) {
        window.location.href = next;
        return;
      }

      // 5. Admin -> /admin
      // 6. Normal user -> /profile
      if (destinationData.isAdmin) {
        window.location.href = "/admin";
        return;
      }

      window.location.href = "/profile";
    } catch (error) {
      console.error("Login error:", error);

      );

  if (!value) return "Not specified";

  return date.toLocaleDateString("en-IN", {
    day: "2-digit",

    year: "numeric",
  });
}
